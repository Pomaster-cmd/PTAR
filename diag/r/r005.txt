from pathlib import Path
root=Path(__file__).resolve().parents[2]
s=(root/'diag/source/win81_nis_proxy.c').read_text(errors='replace')
r=(root/'diag/source/modules/fg_devicec_reconstruction.inc').read_text(errors='replace')
c=(root/'diag/source/modules/fg_custom_edge_lite_b18a.inc').read_text(errors='replace')
checks=[]
def ck(n,v): checks.append((n,bool(v)))
# Locked P1U46 NVIDIA NIS remains available outside frame generation.
ck('P1U46 horizontal NIS shader present','static const char g_nisHorizontalCS[]' in s)
ck('P1U46 vertical NIS shader present','static const char g_nisVerticalCS[]' in s)
ck('P1U46 fused NIS output shader present','static const char g_nisFusedSharpenCopyShaderPS[]' in s)
ck('P1U46 NIS coefficient table reused','init.pSysMem=g_nisCoef64x8' in s)
ck('P1U46 NIS resource builder present','static HRESULT NISEnsureResources' in s)
ck('P1U46 true NIS dispatcher present','static HRESULT NISDispatch(' in s)
ck('P1U46 NIS25 profile still exposed','NVIDIA NIS 25%' in s)
# B18A FG policy: custom reconstruction only, no call to true FG NIS.
ck('B18A custom shader module included','#include "modules/fg_custom_edge_lite_b18a.inc"' in s)
ck('B18A custom PS compiled','CompileBlob(g_fgCustomEdgeLiteShaderPS,"main","ps_5_0"' in s)
ck('FG renderer contains no FGNISDispatch call','FGNISDispatch(' not in r)
ck('FG renderer chooses custom PS','g_fgCustomEdgeLitePS' in r)
ck('FG renderer fail-opens to bilinear','g_fgPresentPSBilinear' in r)
ck('custom kernel has exactly five loads',c.count('Src.Load')==5)
ck('custom kernel has no UAV','RWTexture' not in c and 'UnorderedAccess' not in c)
ck('telemetry reports true NIS FG dispatch count','TRUE NVIDIA NIS FG DISPATCHES' in s)
print(f'B18A NIS POLICY PARITY: {sum(v for _,v in checks)}/{len(checks)} PASS')
for n,v in checks: print(('PASS ' if v else 'FAIL ')+n)
raise SystemExit(0 if all(v for _,v in checks) else 1)
