from pathlib import Path
import re
root=Path(__file__).resolve().parents[2]
new=(root/'diag/source/win81_nis_proxy.c').read_text(errors='replace')
old=(root/'diag/source/win81_nis_proxy.c.vblank2.reference').read_text(errors='replace')
checks=[]
def ck(n,c):checks.append((n,bool(c)))
def funcs(src):
    out={};pat=re.compile(r'(?m)^static\s+(?:__declspec\([^\n]+\)\s+)?[^;\n=]+?\s+([A-Za-z_]\w*)\s*\([^;\n]*\)\s*\{')
    for m in pat.finditer(src):
        i=m.end()-1;d=0;j=i;ins=False;esc=False;q=''
        while j<len(src):
            c=src[j]
            if ins:
                if esc:esc=False
                elif c=='\\':esc=True
                elif c==q:ins=False
            else:
                if c in ('"',"'"):ins=True;q=c
                elif c=='{':d+=1
                elif c=='}':
                    d-=1
                    if d==0:out[m.group(1)]=src[m.start():j+1];break
            j+=1
    return out
of,nf=funcs(old),funcs(new)
# Existing functions allowed to change. FGEnsureResources/PollHotkeys are labels only.
allowed={'StartLogSession','FGDiagResetSessionStats','FGDiagResetActivationState','FGReleaseAll','FGCompilePresenterPipeline','FGRenderFrameMailbox','FGEnsureResources','PollHotkeys','USRLogStatus'}
new_allowed={'FGNISReleaseSizeResources','FGNISReleaseAllResources','FGNISBuildConfig','FGNISCreateTexturePair','FGNISEnsureResources','FGNISDispatch'}
changed={n for n in set(of)&set(nf) if of[n]!=nf[n]}
added=set(nf)-set(of)
ck('changed existing scope exact',changed==allowed)
ck('new helper scope exact',added==new_allowed)
# Validated temporal/VBlank core must remain byte-identical.
critical=['FGDisplayProc','FGPublisherProc','FGPublisherQueueJob','FGPublisherQueueRealOnly','FGCadenceWaitNextTick','FGSubmitFrame','FGPollCompletedJobs','FGNvencBurstObserve','FGPressureObserve','FGPressureProtectSource','FGLoadShedObserveLatency','FGLoadShedConsume','FGLoadShedArm','FGLoadShedEnsure','FGAsyncSubmitFromSlot','FGDispatchInterpolationJob','FGDownscaleOne','FGWorkerProc','HotkeyActionEdge','FGVblankWaitDisplaySlot','FGDisplayPresentBackbuffer','FGMailboxPublishFrame']
for name in critical:ck('unchanged '+name,name in of and name in nf and of[name]==nf[name])
def between(src,a,b):
    i=src.index(a);j=src.index(b,i);return src[i:j]
ck('unchanged FGDisplayTakeMailbox',between(new,'static int FGDisplayTakeMailbox(void){','static int FGVblankWaitDisplaySlot(void){')==between(old,'static int FGDisplayTakeMailbox(void){','static int FGVblankWaitDisplaySlot(void){'))
# No new synchronization/topology/present mechanisms.
for tok in ['CreateThread(','Sleep(','WaitForSingleObject(','Flush(','Present(','D3D11CreateDevice','DXGI_SWAP_EFFECT_FLIP_SEQUENTIAL','FRAME_LATENCY_WAITABLE_OBJECT','DwmFlush']:
    ck('no new '+tok,new.count(tok)<=old.count(tok))
ck('exact N present retained','present?present(g_fgSwap,0u,DXGI_PRESENT_DO_NOT_WAIT)' in new)
ck('VBlank wait retained','hr=waitFn(g_fgVblankOutput)' in new)
# NIS must live on the existing FG device/context and feed the mailbox renderer.
for tok in ['g_fgNisCSHorizontal','g_fgNisCSVertical','g_fgNisScaledSRV','FGNISDispatch(sourceSRV,sourceW,sourceH,fitW,fitH,g_sharpPct)','g_fgDevice','g_fgCtx']:
    ck('combined token '+tok,tok in new)
ck('generated dispatch telemetry','g_fgNisDispatchGenerated++' in new)
ck('real dispatch telemetry','g_fgNisDispatchReal++' in new)
ck('bilinear fail-open','g_fgNisFallbackGenerated++' in new and 'g_fgNisFallbackReal++' in new and 'BILINEAR fail-open active' in new)
ck('resource failure latches fail-open','g_fgNisResourceFailures++;g_fgNisAvailable=0;FGNISReleaseAllResources()' in new)
ck('NIS timing gated by diagnostics','g_vblankDiagnostics&&g_qpcFreq' in nf['FGRenderFrameMailbox'])
ck('NIS compilation selected profile only','g_usrScalerMode==USR_SCALER_SHARP&&g_fgNisAvailable' in nf['FGCompilePresenterPipeline'])
ck('native bypass explicit','else if(g_usrScalerMode==USR_SCALER_SHARP){g_fgNisBypassFrames++;}' in nf['FGRenderFrameMailbox'])
print('CHANGED='+','.join(sorted(changed)))
print('ADDED='+','.join(sorted(added)))
print(f'NISFG1 SCOPE AUDIT: {sum(v for _,v in checks)}/{len(checks)} PASS')
for n,v in checks:print(('PASS ' if v else 'FAIL ')+n)
raise SystemExit(1 if not all(v for _,v in checks) else 0)
